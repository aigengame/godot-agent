def rebind(x):
 if isinstance(x,dict):
  y={k:rebind(v) for k,v in x.items()}
  owner=x.get('package',x.get('id'))
  if isinstance(owner,str) and owner in VERSIONS and 'version' in y: y['version']=VERSIONS[owner]
  if isinstance(x.get('package_id'),str) and x.get('package_id') in VERSIONS and 'package_version' in y: y['package_version']=VERSIONS[x['package_id']]
  return y
 if isinstance(x,list):return [rebind(v) for v in x]
 if isinstance(x,str):
  if x=='quantity.dimensionless-int64':x='quantity.dimensionless-int64-v2-2'
  for a,b in COORDS.items(): x=x.replace(a,b)
  return x
 return x

def rebind_package(p):
 y=rebind(p)
 for closure in y['semantic_closure']:
  for definition in closure['definitions']:
   if isinstance(definition,dict) and 'version' in definition:definition['version']=VERSIONS[p['id']]
 return y

def key(x):
 if isinstance(x,dict):
  for k in ('authority_path','id','code'): 
   if k in x:return (k,x[k])
 return json.dumps(x,sort_keys=True)

conflicts=[]
def union(a,b,path):
 if a==b:return deepcopy(a)
 if isinstance(a,dict) and isinstance(b,dict):
  return {k:union(a[k],b[k],path+'/'+k) if k in a and k in b else deepcopy(a.get(k,b.get(k))) for k in sorted(set(a)|set(b))}
 if isinstance(a,list) and isinstance(b,list):
  y=deepcopy(a); positions={key(v):i for i,v in enumerate(y)}
  for x in b:
   if key(x) in positions:
    i=positions[key(x)]
    y[i]=union(y[i],x,path+'/'+str(key(x)))
   else: positions[key(x)]=len(y);y.append(deepcopy(x))
  return y
 if path.rsplit('/',1)[-1] in ('content_identity','semantic_identity','byte_size'): return deepcopy(a)
 conflicts.append({'path':path,'selected':a,'old':b})
 return deepcopy(a)

def dump(p,x):p.parent.mkdir(parents=True,exist_ok=True);p.write_text(json.dumps(deepcopy(x),indent=2,sort_keys=True)+'\n')

def refresh_embedded_artifacts(target):
 files={p:json.loads(p.read_text()) for p in (target/'packages').rglob('*.json')}
 contracts={};embedded=[]
 def walk(x):
  if isinstance(x,dict):
   if 'artifact_kind' in x and 'identity_domain' in x and 'identity_excluded_members' in x:contracts[x['artifact_kind']]=x
   if isinstance(x.get('const'),dict) and 'content_identity' in x['const'] and 'artifact_kind' in x['const']:embedded.append(x['const'])
   for v in x.values():walk(v)
  elif isinstance(x,list):
   for v in x:walk(v)
 for d in files.values():walk(d)
 replacements={}
 for d in embedded:
  c=contracts[d['artifact_kind']];excluded=set(c['identity_excluded_members'])|{'content_identity'}
  new=content_identity(c['identity_domain'],{k:v for k,v in d.items() if k not in excluded})
  replacements[d['content_identity']]=new
 def replace(x):
  if isinstance(x,dict):return {k:replace(v) for k,v in x.items()}
  if isinstance(x,list):return [replace(v) for v in x]
  return replacements.get(x,x) if isinstance(x,str) else x
 for p,d in files.items():dump(p,replace(d))
