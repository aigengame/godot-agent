ROOT=Path(os.environ['GDA_PROBE_ROOT'])

mode=sys.argv[1]
sys.path.insert(0,str((OUT if mode=='prepared' else ROOT)/'src'))
sys.path.insert(1,str(ROOT/'tests'))
from gda_balancing.domain.model import _checking as checking, _compilation as compilation, _lowering as lowering, _admission as admission

ctx=packaged_authority_context()
counts=Counter(); projection_steps=[]
functions=[lowering.lowering_inputs,lowering._resolved_formulas_and_bindings,lowering._runtime_projection,lowering._resolved_entrypoints,lowering._resolved_call_sites,admission.admit_resolved_model]
for original in functions:
    @functools.wraps(original)
    def wrapper(*args,_fn=original,**kwargs):
        name=_fn.__name__; counts[name]+=1
        try: return _fn(*args,**kwargs)
        finally:
            if name=='_runtime_projection': projection_steps.append({'limit':args[3].limit,'used':args[3].used})
    for name,module in list(sys.modules.items()):
        if module is not None and name.startswith('gda_balancing'):
            for attr,value in list(vars(module).items()):
                if value is original: setattr(module,attr,wrapper)

def reset(): counts.clear();projection_steps.clear()
def observations(): return {'calls':dict(counts),'runtime_projection_steps':list(projection_steps)}
def refusal(value):
    assert isinstance(value,Schema2RefusalReport), repr(type(value))
    return value.model_dump(mode='json')
def compile_source(source):
    checked=checking.check_model_source_value(source,authority_context=ctx)
    assert isinstance(checked,CheckedModel), repr(checked)
    return checked,compilation.compile_checked_model(checked)

    encoded=canonical_bytes(artifacts)
    (OUT/f'{mode}-{path.parent.name}.json').write_bytes(encoded)
    result['examples'][path.parent.name]={'sha256':hashlib.sha256(encoded).hexdigest(),'bytes':len(encoded),**observations()}
    before=canonical_bytes(artifacts)

    imported['rir-semantic-payload']['declarations'][0]['role']='CORRUPTED'

    result['examples'][path.parent.name]['independent_import_admission_unchanged_and_corruption_refused']=True

        artifacts['rir-semantic-payload']['declarations'][0]['role']='CORRUPTED'

        result['examples'][path.parent.name]['prepared_and_output_mutation_isolation']=True
