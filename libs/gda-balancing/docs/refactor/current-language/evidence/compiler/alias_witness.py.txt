def diff(a,b,path=''):
    if a==b:return
    if isinstance(a,dict) and isinstance(b,dict):
        for k in sorted(a.keys()|b.keys()):
            if k not in a or k not in b:result['differences'].append({'path':path+'/'+k,'original':a.get(k),'detached':b.get(k)})
            else:diff(a[k],b[k],path+'/'+k)
    elif isinstance(a,list) and isinstance(b,list) and len(a)==len(b):
        for i,(x,y) in enumerate(zip(a,b)):diff(x,y,path+'/'+str(i))
    else:result['differences'].append({'path':path,'original':a,'detached':b})
